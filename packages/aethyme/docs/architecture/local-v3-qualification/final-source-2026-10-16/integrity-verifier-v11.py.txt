def fail(label, message):
    errors.append(f"{label}: {message}")


def path_parts(value, label):
    if not isinstance(value, str) or not value:
        raise ValueError(f"{label}: empty or non-string path")
    if "\x00" in value or "\\" in value or value.startswith("/"):
        raise ValueError(f"{label}: non-canonical or absolute path {value!r}")
    if re.match(r"\A[A-Za-z]:", value):
        raise ValueError(f"{label}: drive-qualified path {value!r}")
    parts = value.split("/")
    if any(part in ("", ".", "..") for part in parts):
        raise ValueError(f"{label}: unsafe path components {value!r}")
    return parts


def base_relative(base, label):
    try:
        rel = base.relative_to(root)
    except ValueError:
        raise ValueError(f"{label}: base escapes root")
    if rel == pathlib.Path("."):
        return []
    parts = list(rel.parts)
    if any(part in ("", ".", "..") for part in parts):
        raise ValueError(f"{label}: unsafe base path")
    return parts


def safe_file(relative, label, max_bytes=None):
    """Return (repo-relative POSIX name, bytes), rejecting symlinks at every step."""
    try:
        parts = path_parts(relative, label)
    except ValueError as exc:
        fail(label, str(exc))
        return None, None
    current = root
    for index, part in enumerate(parts):
        current = current / part
        try:
            info = current.lstat()
        except OSError as exc:
            fail(label, f"missing/unreadable {relative}: {exc}")
            return None, None
        if stat.S_ISLNK(info.st_mode):
            fail(label, f"symlink path component {relative}")
            return None, None
        if index < len(parts) - 1 and not stat.S_ISDIR(info.st_mode):
            fail(label, f"non-directory path component {relative}")
            return None, None
        if index == len(parts) - 1 and not stat.S_ISREG(info.st_mode):
            fail(label, f"not a regular file {relative}")
            return None, None
    normalized = "/".join(parts)
    if normalized not in tracked:
        fail(label, f"untracked {normalized}")
    if max_bytes is not None and info.st_size > max_bytes:
        fail(label, f"file exceeds {max_bytes}-byte read limit: {normalized}")
        return normalized, None
    try:
        if max_bytes is None:
            data = current.read_bytes()
        else:
            with current.open("rb") as stream:
                data = stream.read(max_bytes + 1)
            if len(data) > max_bytes:
                fail(label, f"file exceeds {max_bytes}-byte read limit: {normalized}")
                return normalized, None
        return normalized, data
    except OSError as exc:
        fail(label, f"cannot read {normalized}: {exc}")
        return normalized, None


def safe_directory(relative, label):
    try:
        parts = path_parts(relative, label)
    except ValueError as exc:
        fail(label, str(exc))
        return None
    current = root
    for part in parts:
        current = current / part
        try:
            info = current.lstat()
        except OSError as exc:
            fail(label, f"missing/unreadable directory {relative}: {exc}")
            return None
        if stat.S_ISLNK(info.st_mode) or not stat.S_ISDIR(info.st_mode):
            fail(label, f"unsafe or non-directory component {relative}")
            return None
    return current


def read_json(relative, label):
    _, data = safe_file(relative, label)
    if data is None:
        return None
    try:
        value = json.loads(data)
    except (json.JSONDecodeError, UnicodeError) as exc:
        fail(label, f"invalid JSON: {exc}")
        return None
    if not isinstance(value, dict):
        fail(label, "expected JSON object")
        return None
    return value


def check(base, rows, label, include_legacy_seen=True, include_group=True):
    """Check a declared list of files relative to a safe repository directory."""
    try:
        prefix = base_relative(base, label)
    except ValueError as exc:
        fail(label, str(exc))
        if include_group:
            groups.append({"inventory": label, "checked": 0})
        return 0
    if not isinstance(rows, list):
        fail(label, "artifact list is missing or not an array")
        if include_group:
            groups.append({"inventory": label, "checked": 0})
        return 0
    count = 0
    local_paths = set()
    for row in rows:
        if not isinstance(row, dict):
            fail(label, "malformed artifact row")
            continue
        name = row.get("path")
        digest = row.get("sha256")
        size = row.get("bytes", row.get("size_bytes"))
        try:
            parts = path_parts(name, label)
        except ValueError as exc:
            fail(label, str(exc))
            continue
        relative = "/".join(prefix + parts)
        if relative in local_paths:
            fail(label, f"duplicate path {relative}")
            continue
        local_paths.add(relative)
        if not isinstance(digest, str) or not HEX64.fullmatch(digest):
            fail(label, f"missing or malformed SHA-256 for {relative}")
            continue
        if size is not None and (not isinstance(size, int) or isinstance(size, bool) or size < 0):
            fail(label, f"invalid byte count for {relative}")
            continue
        _, data = safe_file(relative, label)
        if data is None:
            continue
        if hashlib.sha256(data).hexdigest() != digest:
            fail(label, f"hash mismatch {relative}")
        if size is not None and len(data) != size:
            fail(label, f"size mismatch {relative}")
        if include_legacy_seen:
            seen.add(relative)
        count += 1
    if include_group:
        groups.append({"inventory": label, "checked": count})
    return count


def require_manifest(relative, label):
    value = read_json(relative, label)
    if value is None:
        return None
    return value


def strict_runtime_json(data, label):
    """Parse the new v1 package with duplicate-key and non-standard-value rejection."""
    def unique_object(pairs):
        result = {}
        for key, value in pairs:
            if key in result:
                raise ValueError(f"{label}: duplicate JSON object key {key!r}")
            result[key] = value
        return result

    def reject_constant(value):
        raise ValueError(f"{label}: non-standard JSON value {value}")

    value = json.loads(
        data.decode("utf-8"), object_pairs_hook=unique_object,
        parse_constant=reject_constant,
    )
    if not isinstance(value, dict):
        raise ValueError(f"{label}: expected JSON object")
    return value

def manifest_rows(data, label):
    rows = []
    names = set()
    for line in data.decode("utf-8").splitlines():
        match = re.fullmatch(r"([0-9a-f]{64})  (.+)", line)
        if not match:
            raise ValueError(f"{label}: malformed digest row")
        name = match.group(2)
        while name.startswith("./"):
            name = name[2:]
        parts = path_parts(name, label)
        name = "/".join(parts)
        if name in names:
            raise ValueError(f"{label}: duplicate path {name}")
        names.add(name)
        rows.append({"path": name, "sha256": match.group(1)})
    if not rows:
        raise ValueError(f"{label}: empty manifest")
    return rows

def safe_link_target(relative_target, label):
    """Resolve a local markdown target without following symlinks or leaving root."""
    if "\x00" in relative_target or "\\" in relative_target or relative_target.startswith("/"):
        fail(label, f"unsafe link target {relative_target!r}")
        return
    if re.match(r"\A[A-Za-z]:", relative_target):
        fail(label, f"drive-qualified link target {relative_target!r}")
        return
    combined = posixpath.normpath(posixpath.join(posixpath.dirname(label), relative_target))
    if combined in ("", ".", "..") or combined.startswith("../") or combined.startswith("/"):
        fail(label, f"link escapes root: {relative_target}")
        return
    current = root
    parts = combined.split("/")
    for index, part in enumerate(parts):
        current = current / part
        try:
            info = current.lstat()
        except OSError as exc:
            fail(label, f"missing link target {relative_target}: {exc}")
            return
        if stat.S_ISLNK(info.st_mode):
            fail(label, f"symlink link target {relative_target}")
            return
        if index < len(parts) - 1 and not stat.S_ISDIR(info.st_mode):
            fail(label, f"non-directory link component {relative_target}")
            return
        if index == len(parts) - 1 and not (stat.S_ISDIR(info.st_mode) or stat.S_ISREG(info.st_mode)):
            fail(label, f"invalid link target {relative_target}")
