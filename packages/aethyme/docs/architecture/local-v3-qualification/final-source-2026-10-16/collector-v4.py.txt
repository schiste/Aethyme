def sha(data):
    return hashlib.sha256(data).hexdigest()

def clean_source():
    def git(*args):
        return subprocess.check_output(['/opt/homebrew/bin/git', *args], cwd=ROOT).decode().strip()
    require(git('rev-parse', 'HEAD') == HEAD, "validation failed: git('rev-parse', 'HEAD') == HEAD")
    require(git('rev-parse', 'HEAD^{tree}') == TREE, "validation failed: git('rev-parse', 'HEAD^{tree}') == TREE")
    require(git('status', '--porcelain=v1', '--untracked-files=all') == '', "validation failed: git('status', '--porcelain=v1', '--untracked-files=all') == ''")

def read_regular(path):
    before = path.lstat()
    require(stat.S_ISREG(before.st_mode) and (not path.is_symlink()), path)
    require(before.st_size <= 32 * 1024 * 1024, path)
    data = path.read_bytes()
    after = path.lstat()
    require((before.st_dev, before.st_ino, before.st_size, before.st_mtime_ns) == (after.st_dev, after.st_ino, after.st_size, after.st_mtime_ns), path)
    require(len(data) == before.st_size, 'validation failed: len(data) == before.st_size')
    return data

def add_payload(name, path):
    global loaded_bytes
    require(name not in payloads, 'duplicate archive member: ' + name)
    size = path.lstat().st_size
    require(loaded_bytes + size <= 64 * 1024 * 1024, 'aggregate payload limit exceeded')
    data = read_regular(path)
    require(loaded_bytes + len(data) <= 64 * 1024 * 1024, 'aggregate payload limit exceeded')
    loaded_bytes += len(data)
    payloads[name] = data

    require(base.is_dir() and (not base.is_symlink()), 'validation failed: base.is_dir() and (not base.is_symlink())')
    rows = []
    for path in sorted(base.rglob('*')):
        st = path.lstat()
        require(not path.is_symlink(), path)
        if stat.S_ISDIR(st.st_mode):
            continue
        rel = path.relative_to(base)
        require(all((part not in ('', '.', '..') for part in rel.parts)), "validation failed: all((part not in ('', '.', '..') for part in rel.parts))")
        name = label + '/' + rel.as_posix()
        require(name not in payloads, 'validation failed: name not in payloads')
        add_payload(name, path)
        rows.append(name)
    require(expected_count is None or len(rows) == expected_count, (label, len(rows)))

                member.size = len(data)
                member.mode = 0o600
                member.mtime = 0
                bundle.addfile(member, io.BytesIO(data))
    raw.flush()
    os.fsync(raw.fileno())
